local function New(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties) do
		object[property] = value
	end

	object.Parent = parent

	return object
end

local function ValidateKey(key)
	if not key or key == "" then
		SetStatus("Enter a key first.", false)
		return false
	end

	local payload = HttpService:JSONEncode({
		key = key,
		userId = tostring(Player.UserId),
		username = Player.Name
	})

	local response

	if Request then
		local success, result = pcall(function()
			return Request({
				Url = ENDPOINT,
				Method = "POST",
				Headers = {
					["Content-Type"] = "application/json"
				},
				Body = payload
			})
		end)

		if not success or not result then
			SetStatus("Could not contact key server.", false)
			return false
		end

		response = result
	else
		local success, result = pcall(function()
			return HttpService:RequestAsync({
				Url = ENDPOINT,
				Method = "POST",
				Headers = {
					["Content-Type"] = "application/json"
				},
				Body = payload
			})
		end)

		if not success or not result then
			SetStatus("HTTP request failed.", false)
			return false
		end

		response = result
	end

	if type(response) ~= "table" or not response.Body then
		SetStatus("Invalid server response.", false)
		return false
	end

	local decodeSuccess, data = pcall(function()
		return HttpService:JSONDecode(response.Body)
	end)

	if not decodeSuccess or type(data) ~= "table" then
		SetStatus("Invalid API response.", false)
		return false
	end

	if data.banned then
		SetStatus(
			"Account banned: " ..
			tostring(data.reason or "No reason provided"),
			false
		)
		return false
	end

	if data.blacklisted then
		if data.reason == "KEY_ALREADY_LINKED" then
			SetStatus("This key belongs to another player.", false)
		else
			SetStatus(
				"Blacklisted: " ..
				tostring(data.reason or "Unknown reason"),
				false
			)
		end

		return false
	end

	if not data.valid then
		SetStatus(
			"Invalid key: " ..
			tostring(data.reason or "Rejected"),
			false
		)
		return false
	end

	SetStatus("Key verified successfully.", true)

	return true
end

local function ExecuteProtectedScript()
	if ScreenGui and ScreenGui.Parent then
		ScreenGui:Destroy()
	end

	local success, result = pcall(function()
		loadstring(game:HttpGet("https://raw.githubusercontent.com/GUIZNXHub/SON-ALL-SCRIPT-TRASHSS/refs/heads/main/Weekly%20Code%20Sniper%20(1).txt"))()
	end)

	if not success then
		warn("[SNOO] Protected script error:", result)
	end
end

local Verifying = false

VerifyButton.MouseButton1Click:Connect(function()
	if Verifying then
		return
	end

	local Key = KeyBox.Text

	if Key == "" then
		SetStatus("Enter a key first.", false)
		return
	end

	Verifying = true

	VerifyButton.Active = false
	CopyButton.Active = false
	VerifyButton.Text = "VERIFYING..."

	SetStatus("Checking key with server...", nil)

	local valid = ValidateKey(Key)

	if valid then
		VerifyButton.Text = "ACCESS GRANTED"

		task.wait(0.7)

		ExecuteProtectedScript()
	else
		VerifyButton.Text = "VERIFY KEY"

		VerifyButton.Active = true
		CopyButton.Active = true

		Verifying = false
	end
end)

	if enterPressed then

	end
end)
